# Payment examples

Use these examples to send AlphaUSD on Moderato testnet. Start with [Send payments](/docs/guide/payments/send-a-payment) for the integration flow, or [Browser payments](/docs/guide/payments/send-a-payment/browser) for a complete React walkthrough.

Fund a test account with the [faucet](/docs/quickstart/faucet) and replace the example recipients with accounts you control. Set `RPC_URL` or `TEMPO_RPC_URL` to `https://rpc.moderato.tempo.xyz` for command-line examples. Use the [SDK guides](/docs/sdk) to install and configure your language's tooling. Cast examples use `TEMPO_PRIVATE_KEY` from the [CLI setup](#cli-client); all token amounts are in base units.

## Send a payment

Send a payment using the standard `transfer` function. The Solidity examples are owner-operated treasury contracts: pass the token address to the constructor and fund the contract before calling it.

    ```go [main.go]
    package main

    import (
    	"context"
    	"log"
    	"math/big"

    	"github.com/ethereum/go-ethereum/common"
    	"github.com/tempoxyz/tempo-go/pkg/signer"
    	"github.com/tempoxyz/tempo-go/pkg/transaction"
    )

    func buildTransferData(to common.Address, amount *big.Int) []byte {
    	data := make([]byte, 68) // 4 (selector) + 32 (address) + 32 (uint256)
    	data[0], data[1], data[2], data[3] = 0xa9, 0x05, 0x9c, 0xbb
    	copy(data[16:36], to.Bytes())
    	amount.FillBytes(data[36:68])
    	return data
    }

    func main() {
    	sgn, _ := signer.NewSigner("0x...")
    	c := newClient()
    	ctx := context.Background()

    	nonce, _ := c.GetTransactionCount(ctx, sgn.Address().Hex())

    	recipient := common.HexToAddress("0x742d35Cc6634C0532925a3b844Bc9e7595f0bEbb")
    	alphaUSD := common.HexToAddress("0x20c0000000000000000000000000000000000001")

        tx := transaction.NewBuilder(big.NewInt(42431)).
    		SetNonce(nonce).
    		SetGas(100_000).
    		SetMaxFeePerGas(big.NewInt(25_000_000_000)).
    		SetMaxPriorityFeePerGas(big.NewInt(1_000_000_000)).
    		AddCall( // [!code hl]
    			alphaUSD, // [!code hl]
    			big.NewInt(0), // [!code hl]
    			buildTransferData(recipient, big.NewInt(100_000_000)), // [!code hl]
    		). // [!code hl]
    		Build()

    	_ = transaction.SignTransaction(tx, sgn)
    	serialized, _ := transaction.Serialize(tx, nil)
    	txHash, _ := c.SendRawTransaction(ctx, serialized)

    	log.Printf("Transaction hash: %s", txHash)
    }
    ```

## Add a payment reference

Include a memo for payment reconciliation and tracking. Use a short invoice or order ID. Follow [Payment references](/docs/guide/payments/transfer-memos#encode-a-payment-reference) for the shared UTF-8 encoding and length checks.

## Batch payments

Send several payments as calls in one transaction. If a call reverts, the whole batch reverts. For separate transaction receipts, [send payments in parallel](/docs/guide/payments/send-parallel-transactions).

<Tabs stateKey="library">
  <Tab title="Viem">

    :::code-group

    ```ts twoslash [example.ts]
    // @noErrors
    import { parseUnits } from 'viem'
    import { Actions } from 'viem/tempo'
    import { client } from './viem.config'

    const token = '0x20c0000000000000000000000000000000000001' as const
    const payments = [
      { to: '0x742d35cc6634c0532925a3b844bc9e7595f0bebb', amount: parseUnits('100', 6) },
      { to: '0x70997970c51812dc3a010c7d01b50e0d17dc79c8', amount: parseUnits('50', 6) },
    ] as const

    const calls = payments.map(({ to, amount }) =>
      Actions.token.transfer.call({ token, to, amount }), // [!code hl]
    )

    await client.sendTransaction({ calls }) // [!code hl]
    ```

    ```rust [example.rs]
    use alloy::{
        primitives::{address, Address, U256},
        providers::{Provider, ProviderBuilder},
        sol_types::SolCall,
    };
    use tempo_alloy::{
        TempoNetwork, contracts::precompiles::ITIP20, primitives::transaction::Call,
        rpc::TempoTransactionRequest,
    };

    mod provider;

    #[tokio::main]
    async fn main() -> Result<(), Box<dyn std::error::Error>> {
        let provider = provider::get_provider().await?;

        let recipient1 = address!("0x742d35Cc6634C0532925a3b844Bc9e7595f0bEbb");
        let recipient2 = address!("0x70997970C51812dc3A010C7d01b50e0d17dc79C8");
        let token_address: Address = address!("0x20c0000000000000000000000000000000000001");

        let calls = vec![ // [!code hl]
            Call { // [!code hl]
                to: token_address.into(), // [!code hl]
                input: ITIP20::transferCall { // [!code hl]
                    to: recipient1, // [!code hl]
                    amount: U256::from(100_000_000), // [!code hl]
                } // [!code hl]
                .abi_encode() // [!code hl]
                .into(), // [!code hl]
                value: U256::ZERO, // [!code hl]
            }, // [!code hl]
            Call { // [!code hl]
                to: token_address.into(), // [!code hl]
                input: ITIP20::transferCall { // [!code hl]
                    to: recipient2, // [!code hl]
                    amount: U256::from(50_000_000), // [!code hl]
                } // [!code hl]
                .abi_encode() // [!code hl]
                .into(), // [!code hl]
                value: U256::ZERO, // [!code hl]
            }, // [!code hl]
        ]; // [!code hl]

        let pending = provider
            .send_transaction(TempoTransactionRequest {
                calls,
                ..Default::default()
            })
            .await?;
        let tx_hash = pending.tx_hash();

        println!("Batch transaction sent: {tx_hash:?}");

        Ok(())
    }
    ```

    ```go [main.go]
    		AddCall(alphaUSD, big.NewInt(0), buildTransferData(recipient1, big.NewInt(100_000_000))). // [!code hl]
    		AddCall(alphaUSD, big.NewInt(0), buildTransferData(recipient2, big.NewInt(50_000_000))). // [!code hl]
    		Build()

    	_ = transaction.SignTransaction(tx, sgn)
    	serialized, _ := transaction.Serialize(tx, nil)
    	txHash, _ := c.SendRawTransaction(ctx, serialized)

    	log.Printf("Batch transaction sent: %s", txHash)
    }
    ```

## Payment client setup

Configure a client for creating accounts, reading balances, and sending payments. These setups supply the shared configuration used in the payment guides. API examples use Moderato with `chainId=42431`.

### Choose your tools

The examples on this page and in the payment guides use AlphaUSD on Moderato. Choose one signing environment for each workflow: a CLI/private-key signer, a Viem client, or a wallet connected through Wagmi. Read-only examples need no private key.

- [CLI](#cli-client): install the current Foundry release with Tempo support.
- [Viem](#viem-client): use a local script or backend signer.
- [Wagmi](#wagmi-client): connect Tempo Wallet in React.
- [Python](#python-client): use the typed Python SDK.

### CLI client

Follow [Foundry installation](/docs/sdk/foundry#get-started-with-foundry). These examples require a current release with Tempo support, including `cast batch-send` and `cast keychain`.

```bash
export TEMPO_RPC_URL=https://rpc.moderato.tempo.xyz
export ALPHA_USD=0x20c0000000000000000000000000000000000001
export TEMPO_ACCOUNT_ADDRESS=your_test_account_address
export TEMPO_RECIPIENT=your_test_recipient_address
export TEMPO_PRIVATE_KEY=your_test_account_private_key
```

Replace the last three values with your own test inputs. `cast wallet new` creates a local key and address. Keep the private key in your signing environment. A read-only balance query only needs the RPC URL, token, and address. Set `PAYMENT_TX_HASH` to a returned transaction hash when inspecting its receipt.

`cast erc20 transfer` and token amounts in `cast send` or `cast batch-send` use base units: `1000000` is one AlphaUSD. `cast erc20 balance` also returns base units. `cast logs --subscribe` uses the Moderato WebSocket endpoint; stop it with Ctrl+C.

Set `alphaUsd` to `0x20c0000000000000000000000000000000000001` and `recipient` to your intended address. Read examples take `account` as an address. The configured connector uses Moderato; keep the connected account on that network.

#### Access-key restrictions

For [granting access](/docs/accounts/keys#grant-access), use the new P-256 `accessKey` from the [access-key example](/docs/accounts/access-keys#authorize-a-limited-key). The connected wallet must control the account that key acts for. The official Account Keychain ABI represents the same one-hour, 10 AlphaUSD, recipient-scoped permission as follows:

```ts
const keyRestrictions = {
  expiry: BigInt(Math.floor(Date.now() / 1000) + 3600),
  enforceLimits: true,
  limits: [{ token: alphaUsd, amount: 10_000_000n, period: 0n }],
  allowAnyCalls: false,
  allowedCalls: [{ target: alphaUsd, selectorRules: [{
    selector: '0xa9059cbb', recipients: [recipient],
  }] }],
} as const
```

The raw ABI names the token allowance `amount`; Viem’s helper names it `limit`. In the [Wagmi authorization call](/docs/accounts/keys#grant-access), `1` identifies the P-256 signature type in `authorizeKey`. Use the actual type of your generated key; secp256k1 is `0`. A connected wallet submits this root-authorized contract call; it does not expose its private key to the application.

### Python client

Install the typed SDK with `python -m pip install pytempo==0.6.1 requests`. Set `TEMPO_PRIVATE_KEY`, `TEMPO_RECIPIENT`, and `TEMPO_GAS_LIMIT` before running a payment. The gas limit is your cap for the calls you submit; estimate for your actual transaction instead of assuming every payment has the same cost.

Save this as `payment_setup.py`. It exposes `w3`, `account`, `private_key`, `recipient`, `alpha_usd`, and `transaction_options()` for the examples. The function refreshes the nonce and fee fields each time you build a transaction.

```python [payment_setup.py]
import os
from web3 import Web3
from pytempo import TempoTransaction
from pytempo.contracts import TIP20

w3 = Web3(Web3.HTTPProvider("https://rpc.moderato.tempo.xyz"))
private_key = os.environ["TEMPO_PRIVATE_KEY"]
account = w3.eth.account.from_key(private_key)
recipient = Web3.to_checksum_address(os.environ["TEMPO_RECIPIENT"])
alpha_usd = "0x20c0000000000000000000000000000000000001"

def transaction_options():
    chain_id = w3.eth.chain_id
    if chain_id != 42431:
        raise ValueError("Expected Moderato testnet")
    gas_price = w3.eth.gas_price
    return {
        "chain_id": chain_id,
        "nonce": w3.eth.get_transaction_count(account.address),
        "gas_limit": int(os.environ["TEMPO_GAS_LIMIT"]),
        "max_fee_per_gas": gas_price * 2,
        "max_priority_fee_per_gas": gas_price,
        "fee_token": alpha_usd,
    }
```

Import the setup before a Python payment example:

```python
from payment_setup import (
    w3, account, private_key, recipient, alpha_usd, transaction_options,
)
from pytempo import TempoTransaction
from pytempo.contracts import TIP20
```

`TIP20.transfer` returns a typed call. `TempoTransaction.create` adds transaction fields; `.sign` signs it; `w3.eth.send_raw_transaction` submits the encoded result. To submit the same signed bytes through cURL instead, print `"0x" + signed_tx.encode().hex()` and set `SIGNED_TRANSACTION` to that value. Do not submit both paths unless you intend to rebroadcast the same transaction.

```ts [create-passkey.ts]
import { Account, WebAuthnP256 } from 'viem/tempo'

export async function createPasskeyAccount() {
```

### Onchain balances and history

Read balances and index account activity directly from the chain. Set `TEMPO_ACCOUNT_ADDRESS` and `TEMPO_FROM_BLOCK` before running `balance.mts`.

```ts [balance.mts]
import { getAddress } from 'viem'
import { tempoModerato } from 'viem/chains'
import { Abis, createClient } from 'viem/tempo'

const accountAddress = process.env.TEMPO_ACCOUNT_ADDRESS
if (!accountAddress) throw new Error('Set TEMPO_ACCOUNT_ADDRESS')

const account = getAddress(accountAddress)
const token = '0x20c0000000000000000000000000000000000001'
const client = createClient({ chain: tempoModerato })

const [balance, { symbol }] = await Promise.all([
  client.token.getBalance({ account, token }),
  client.token.getMetadata({ token }),
])

console.log(`${balance.formatted} ${symbol}`)

const fromBlock = process.env.TEMPO_FROM_BLOCK
if (!fromBlock) throw new Error('Set TEMPO_FROM_BLOCK')

const payments = await client.getContractEvents({
  address: token,
  abi: Abis.tip20,
  eventName: 'Transfer',
  args: { to: account },
  fromBlock: BigInt(fromBlock),
  toBlock: 'latest',
})

console.log(payments.map((payment) => ({
  from: payment.args.from,
  amount: payment.args.amount,
  transactionHash: payment.transactionHash,
  logIndex: payment.logIndex,
})))

const unwatch = client.token.watchTransfer({
  token,
  args: { to: account },
  onTransfer: async (transfer, log) => {
    const balance = await client.token.getBalance({ account, token })
    console.log({
      balance: balance.formatted,
      received: transfer.amount,
      transactionHash: log.transactionHash,
      logIndex: log.logIndex,
    })
  },
})

// Call unwatch() when the application no longer needs the subscription.
```

Call `unwatch()` when the subscription is no longer needed. For indexed reads from any language, use the [interactive API tools](/docs/guide/payments/accept-a-payment#read-balances).
